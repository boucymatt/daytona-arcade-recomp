cmake_minimum_required(VERSION 3.20)
project(xbox_fp_probe LANGUAGES C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(SOFTFLOAT_DIR "${ROOT}/extern/softfloat")
include("${ROOT}/cmake/softfloat_sources.cmake")
add_library(probe_softfloat STATIC ${SOFTFLOAT_SOURCES})
target_include_directories(probe_softfloat PUBLIC "${SOFTFLOAT_DIR}/source/include"
    PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${SOFTFLOAT_DIR}/source/8086-SSE")
target_compile_definitions(probe_softfloat PUBLIC SOFTFLOAT_FAST_INT64
    $<$<COMPILE_LANGUAGE:C>:THREAD_LOCAL=_Thread_local>
    $<$<COMPILE_LANGUAGE:CXX>:THREAD_LOCAL=__thread>)
target_compile_definitions(probe_softfloat PRIVATE SOFTFLOAT_ROUND_ODD
    SOFTFLOAT_FAST_DIV32TO16 SOFTFLOAT_FAST_DIV64TO32)
if(CMAKE_C_BYTE_ORDER STREQUAL "LITTLE_ENDIAN")
    target_compile_definitions(probe_softfloat PUBLIC LITTLEENDIAN=1)
elseif(NOT CMAKE_C_BYTE_ORDER STREQUAL "BIG_ENDIAN")
    message(FATAL_ERROR "Unknown host byte order")
endif()
if(XBOX_TARGET AND NOT CMAKE_C_BYTE_ORDER STREQUAL "BIG_ENDIAN")
    message(FATAL_ERROR "Xbox target must use big-endian SoftFloat")
endif()
add_library(probe_fp STATIC "${ROOT}/src/i960/fp.cpp" ../fp_cases.cpp)
target_include_directories(probe_fp PUBLIC "${ROOT}/src" ..)
target_compile_features(probe_fp PUBLIC cxx_std_20)
target_link_libraries(probe_fp PUBLIC probe_softfloat)
foreach(target probe_fp probe_softfloat)
    target_compile_options(${target} PRIVATE -fno-fast-math -ffp-contract=off)
endforeach()
if(NOT XBOX_TARGET)
    add_executable(fp_oracle ../fp_oracle.cpp)
    target_link_libraries(fp_oracle PRIVATE probe_fp)
endif()
